OBJECT: 
 STRING STRF,
 STRING S1,
 STRING S2,
 STRING S3,
 STRING CDPAGE,
 STRING EXPORTFILENAME,
 STRING PICT,
 INTEGER ROWNUM,
 DATE TMPPLANSTART;

/* --- Ayarlar --- */
PICT = '#,###,###,###,###,##0.00';
CDPAGE = 'ISO8859-9';
/* --- Dosya yolları --- */
S1 = 'IASREPORTS/prjt05gant.xls';
S2 = 'IASREPORTS/TMPPRJT.xls';
COPYFILE S1 INTO S2;
SORT IASPRJOPERATION ON ASC PARENTID;
READ IASPRJOPERATION WITH FIRST;
TMPPLANSTART = GETDATE(IASPRJOPERATION_PLANSTART);
ROWNUM = 18;

ACCESSXLS S2 WRITE SHEET 1 C 13 VALUE TMPPLANSTART;

LOOP AT IASPRJOPERATION 
BEGIN
	LOCATERECORD SEQUENTIAL COLUMNS GTSTEXT VALUES IASPRJOPERATION_MATERIAL ON CGNTTABLE;

	IF SYS_STATUS == 0 THEN
		ACCESSXLS S2 WRITE SHEET 1 A ROWNUM VALUE CGNTTABLE_GTLTEXT;
	ENDIF;

	ACCESSXLS S2 WRITE SHEET 1 B ROWNUM VALUE IASPRJOPERATION_TARGETSTART;
	ACCESSXLS S2 WRITE SHEET 1 C ROWNUM VALUE IASPRJOPERATION_TARGETEND;
	ACCESSXLS S2 WRITE SHEET 1 D ROWNUM VALUE IASPRJOPERATION_PLANSTART;
	ACCESSXLS S2 WRITE SHEET 1 E ROWNUM VALUE IASPRJOPERATION_PLANEND;
	ROWNUM = ROWNUM + 1;
	LOCATERECORD SEQUENTIAL COLUMNS GTSTEXT VALUES IASPRJOPERATION_MATERIAL ON CGNTTABLE;

	IF SYS_STATUS == 0 THEN
		ACCESSXLS S2 WRITE SHEET 1 A ROWNUM VALUE CGNTTABLE_GTLTEXT;
	ENDIF;

	ACCESSXLS S2 WRITE SHEET 1 B ROWNUM VALUE IASPRJOPERATION_TARGETSTART;
	ACCESSXLS S2 WRITE SHEET 1 C ROWNUM VALUE IASPRJOPERATION_TARGETEND;
	ACCESSXLS S2 WRITE SHEET 1 D ROWNUM VALUE IASPRJOPERATION_PLANSTART;
	ACCESSXLS S2 WRITE SHEET 1 E ROWNUM VALUE IASPRJOPERATION_PLANEND;
/* Sonraki malzeme için satırı 2 arttır */
	ROWNUM = ROWNUM + 1;
ENDLOOP;

CLOSE FILE;
EXPORTFILENAME = 'C:\TMP\GANTPRJ.xls';
S3 = '*' + EXPORTFILENAME;
COPYFILE S2 INTO S3;
S3 = '*RunDLL32.EXE shell32.dll,ShellExec_RunDLL "' + EXPORTFILENAME + '"';
RUNPROGRAM S3;
