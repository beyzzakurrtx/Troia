OBJECT: 
 STRING PREVRW,
 STRING CURRW,
 STRING PREVCONF,
 STRING CURCONF,
 STRING PREVCONFPOS,
 STRING CURCONFPOS,
 INTEGER SUMQTY,
 DECIMAL SUMREWQUAN,
 DECIMAL SUMONAYMIKTAR,
 DECIMAL SUMTOTAL,
 DECIMAL SUMREWORKRATE,
 DECIMAL TOTALR,
 DECIMAL TOTALRO,
 DECIMAL TOTAL,
 DECIMAL TOTALREWORKRATETMP,
 DECIMAL TMPREWQUAN;

/*HT 00000001 LIVBEYZA SUMONAY MIKTAR DA KALDIM - 02.06.2025*/
SORT REPORTLST ON REWORKKEY;
SELECT  '' AS POTYPE, '' AS PRDORDER, '' AS WORKCENTER, '' AS OPRMATERIAL,
	 '' AS REWORKKEY , '' AS REWORKTEXT , 0.0 AS REWQUAN , '' AS REWORKRATE ,
	 0.0 AS ONAYMIKTAR , '' AS FROMQLT , '' AS CONFIRMATION , '' AS CONFIRMPOS ,
	 0.0 AS TOTALQTY , '' AS OPRMATERIALSTEXT , '01.01.1975' AS CONFIRMDATE  , '' AS TOTALREWORKRATE 
	FROM IASBAS001 
	WHERE 1 = 2 
	INTO REPORTTABLETMP;

/*COPY TABLE REPORTLST INTO REPORTTABLETMP;*/
SORT REPORTLST ON REWORKKEY;

LOOP AT REPORTLST CRITERIA COLUMNS SELECTED VALUES 1 
BEGIN
	LOCATERECORD SEQUENTIAL COLUMNS CONFIRMATION,CONFIRMPOS,REWORKKEY VALUES REPORTLST_CONFIRMATION, REPORTLST_CONFIRMPOS, REPORTLST_REWORKKEY ON  REPORTTABLETMP;

	IF SYS_STATUS == 1 THEN
		APPEND ROW TO REPORTTABLETMP;
		MOVE-CORRESPONDING REPORTLST TO REPORTTABLETMP ;
	ELSE
		SUMONAYMIKTAR = SUMONAYMIKTAR + REPORTTABLETMP_ONAYMIKTAR;
		SUMTOTAL = SUMTOTAL + REPORTTABLETMP_TOTALQTY;
	ENDIF;

ENDLOOP;

DUMP REPORTTABLETMP;
PREVRW = '';
CURRW = '';
SUMONAYMIKTAR= 0;
SUMTOTAL = 0;

LOOP AT REPORTTABLETMP 
BEGIN
	PREVRW = REPORTTABLETMP_REWORKKEY;

	IF PREVRW != CURRW 
			&& CURRW != '' THEN
/*SARI BAŞLANGIÇ*/
		APPEND ROW TO REPORTTABLE;
		SETBACKCOLOR REPORTTABLE TO YELLOW;
		MOVE SUMREWQUAN TO  REPORTTABLE_REWQUAN;
		MOVE SUMONAYMIKTAR TO REPORTTABLE_ONAYMIKTAR;
		SUMREWORKRATE = SUMREWQUAN / (SUMREWQUAN + SUMONAYMIKTAR) * 100 ;
		SUMREWORKRATE = ROUND(SUMREWORKRATE,2);
		MOVE SUMREWORKRATE TO REPORTTABLE_REWORKRATE;
		MOVE SUMTOTAL TO REPORTTABLE_TOTALQTY;
		TOTALR = TOTALR + SUMREWQUAN;
		TOTALRO = TOTALRO + SUMTOTAL;
		SUMREWQUAN = 0;
		SUMONAYMIKTAR = 0;
		SUMREWORKRATE = 0;
		SUMTOTAL = 0;
/*SARI BİTİŞ*/
		APPEND ROW TO REPORTTABLE;
		MOVE-CORRESPONDING REPORTTABLETMP TO REPORTTABLE ;
		SUMREWQUAN = SUMREWQUAN + REPORTTABLETMP_REWQUAN;
		SUMONAYMIKTAR = SUMONAYMIKTAR + REPORTTABLETMP_ONAYMIKTAR;
		SUMTOTAL = SUMTOTAL + REPORTTABLETMP_TOTALQTY;
	ELSE
		SUMONAYMIKTAR = SUMONAYMIKTAR + REPORTTABLETMP_ONAYMIKTAR;
		SUMREWQUAN = SUMREWQUAN + REPORTTABLETMP_REWQUAN;
		SUMTOTAL = SUMTOTAL + REPORTTABLETMP_TOTALQTY;
		APPEND ROW TO REPORTTABLE;
		MOVE-CORRESPONDING REPORTTABLETMP TO REPORTTABLE ;
	ENDIF;

	CURRW = PREVRW;
ENDLOOP;

/*SARI SON SATIR*/
IF PREVRW != '' THEN
	APPEND ROW TO REPORTTABLE;
	SETBACKCOLOR REPORTTABLE TO YELLOW;
	MOVE SUMREWQUAN TO  REPORTTABLE_REWQUAN;
	MOVE SUMONAYMIKTAR TO REPORTTABLE_ONAYMIKTAR;
	SUMREWORKRATE = SUMREWQUAN / (SUMREWQUAN + SUMONAYMIKTAR) * 100 ;
	SUMREWORKRATE = ROUND(SUMREWORKRATE,2);
	MOVE SUMREWORKRATE TO REPORTTABLE_REWORKRATE;
	MOVE SUMTOTAL TO REPORTTABLE_TOTALQTY;
	TOTALR = TOTALR + SUMREWQUAN;
	TOTALRO = TOTALRO + SUMTOTAL;
ENDIF;

TOTAL = (TOTALR / TOTALRO) * 100;
TOTALREQQUAN = TOTALR;
TOTALREQONAY = TOTALRO;
TOTALREQRATE = TOTAL;

LOOP AT REPORTTABLE 
BEGIN
TMPREWQUAN = REPORTTABLE_REWQUAN;
	TOTALREWORKRATETMP = (TMPREWQUAN / TOTALRO) * 100;
	TOTALREWORKRATETMP = ROUND(TOTALREWORKRATETMP,2);
	MOVE TOTALREWORKRATETMP TO REPORTTABLE_TOTALREWORKRATE;
ENDLOOP;
