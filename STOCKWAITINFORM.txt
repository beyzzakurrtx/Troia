OBJECT: 
 DATE DATE1,
 STRING TXTMAILMESSAGE,
 STRING TXTMADRES,
 STRING HTMLROWS,
 STRING CMAT;

TXTMAILMESSAGE = '';
TXTMADRES = '';
HTMLROWS = '';

DATE1 = GETDATE(ADDYEARS(SYS_CURRENTDATE,-1)); /*şuanki tarihden bir yıl önceki date değerini almak için yazıldı.*/

SELECT '' AS MATERIAL , '' AS BATCHNUM , '01.01.1975' DOCDATE 
	FROM IASBAS001 
	WHERE 1=2 
	INTO TMPREPORT; /*boş bir tablo oluşturmak için yazıldı verilerimizi burada toplayacağız.*/

SELECT MATERIAL, WAREHOUSE, STOCKPLACE, BATCHNUM 
	FROM IASINVSTOCK 
	WHERE CLIENT = '00' 
		AND COMPANY = '01' 
		AND PLANT = '01' 
		AND LEN(MATERIAL) = 16 
		AND WAREHOUSE = 'SVK' 
		AND STOCKPLACE LIKE 'P%' 
	INTO CLKINVSTOCKTMP;
	
	/*Yapmış olduğumuz sistemde sevkiyat için bekleyen parçalar SVK depoda ve P kodu ile başlayan stok yerinde gibi bir kurgu olduğu için sorgumuzu sisteme göre yazıyoruz.*/
    /*Siz kendi verilerinize göre düzenleyebilirsiniz.*/

LOOP AT CLKINVSTOCKTMP 
BEGIN
	SELECT TOP 1 I.DOCDATE 
		FROM IASINVITEM I 
		INNER JOIN IASINVHEAD H ON H.CLIENT = I.CLIENT 
			AND H.COMPANY = I.COMPANY 
			AND H.INVDOCTYPE = I.INVDOCTYPE 
			AND H.INVDOCNUM = I.INVDOCNUM 
		WHERE I.CLIENT = SYS_CLIENT 
			AND I.COMPANY = '01' 
			AND I.PLANT = '01' 
			AND I.MATERIAL = CLKINVSTOCKTMP_MATERIAL 
			AND I.BATCHNUM = CLKINVSTOCKTMP_BATCHNUM 
			AND I.QPOSTWAY = 0 
			AND I.WAREHOUSE = CLKINVSTOCKTMP_WAREHOUSE 
			AND I.STOCKPLACE = CLKINVSTOCKTMP_STOCKPLACE 
			AND I.DOCDATE < DATE1 
			AND H.ISCANCELED = 0 
		ORDERBY I.DOCDATE ASC 
		INTO CLKINVTMP;


	IF SELECTED THEN

		IF TMPREPORT_MATERIAL != '' THEN
			APPEND ROW TO TMPREPORT;
		ENDIF;

		MOVE-CORRESPONDING CLKINVSTOCKTMP TO TMPREPORT;
		MOVE CLKINVTMP_DOCDATE  TO TMPREPORT_DOCDATE;
	ENDIF;

ENDLOOP;


IF TMPREPORT_ROWCOUNT > 0 THEN

/*SENDMAIL fonksiyonunda kullanacağımız verileri alıyoruz destek tablosuna doğru şekilde select atarak. Manuel de değer atayabilirsiniz.*/

	SELECT ADDRESS, OUSERNAME, OHOST, OPAWSWORD 
		FROM IASCLB011 
		WHERE CLIENT = SYS_CLIENT 
			AND ADDRESS = 'bilgi@company.com' 
		INTO CLKCLB011TMP;
		
/*Destek tabloları içinde BPM modulünde yapmış olduğumuz maıl şablonunu buluyoruz.*/

	SELECT  * 
		FROM IASBPM012 
		WHERE CLIENT = SYS_CLIENT 
			AND COMPANY = '01' 
			AND TEXTNUM = '147' 
			AND LANGU = SYS_LANGU 
		INTO CLKBPM012TMP;

	SELECT CMATERIAL,CDRAWNUM 
		FROM IASMATCUSTOMERS 
		WHERE CLIENT = SYS_CLIENT 
			AND COMPANY = '01' 
			AND MATERIAL = TMPREPORT_MATERIAL 
			AND VALIDFROM <= SYS_CURRENTDATE 
			AND VALIDUNTIL >= SYS_CURRENTDATE 
		INTO TMPMATC;


	IF TMPMATC_CDRAWNUM!='' THEN
		CMAT = TMPMATC_CDRAWNUM;
	ELSE
		CMAT = TMPMATC_CMATERIAL;
	ENDIF;

	HTMLROWS = '';
	TXTMAILMESSAGE = CLKBPM012TMP_LTEXT;

	LOOP AT TMPREPORT 
	BEGIN
		HTMLROWS = HTMLROWS + '<tr>' + '<td>' + TMPREPORT_MATERIAL + '</td>' + '<td>' + CMAT + '</td>' + '<td>' + TMPREPORT_BATCHNUM + '</td>' + '<td>' + TMPREPORT_DOCDATE + '</td>' + '</tr>';
	ENDLOOP;

	TXTMAILMESSAGE = REPLACE(TXTMAILMESSAGE, '#TABLEROWS#', HTMLROWS);
	TXTMADRES = 'kalite.guvence@company.com,bilgi.islem@company.com';
	SENDMAIL MESSAGE TXTMAILMESSAGE TOADDRESS TXTMADRES HOST CLKCLB011TMP_OHOST PASSW CLKCLB011TMP_OPAWSWORD USERNAME CLKCLB011TMP_OUSERNAME FROM CLKCLB011TMP_ADDRESS SUBJECT CLKBPM012TMP_SUBJECT ENCPRO 'TLS';
ENDIF;
