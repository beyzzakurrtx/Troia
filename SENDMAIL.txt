OBJECT: 
 STRING TXTMAILMESSAGE,
 STRING TXTMADRES ;

SELECT ADDRESS, OUSERNAME, OHOST, OPAWSWORD 
	FROM IASCLB011 
	WHERE CLIENT = SYS_CLIENT 
		AND ADDRESS = 'XXX@gmail.com' 
	INTO CLKCLB011TMP;

SELECT  * 
	FROM IASBPM012 
	WHERE CLIENT = SYS_CLIENT 
		AND COMPANY = '01' 
		AND TEXTNUM = '146' 
		AND LANGU = SYS_LANGU 
	INTO CLKBPM012TMP;

TXTMAILMESSAGE = CLKBPM012TMP_LTEXT;
TXTMAILMESSAGE = REPLACE (TXTMAILMESSAGE,'#CLKUSER#',CLKBARPRINTLOG_USERNAME);
TXTMAILMESSAGE = REPLACE (TXTMAILMESSAGE,'#CLKUSERPASS#',CLKBARPRINTLOG_USERPASS);
TXTMAILMESSAGE = REPLACE (TXTMAILMESSAGE,'#CLKREQUESTTIME#',CLKBARPRINTLOG_REQUESTTIME);
TXTMAILMESSAGE = REPLACE (TXTMAILMESSAGE,'#CLKBATCHNUM#',CLKBARPRINTLOG_BATCHNUM);
TXTMAILMESSAGE = '<html><head><meta http-equiv="Content-Type" content="text/html; charset=UTF-8"></head><body>' + TXTMAILMESSAGE + '</body></html>';
LOCAL: 
 STRING DBNAME;

DBNAME=GETUSERINFO('database_name');

IF DBNAME == 'TEST803' THEN
	TXTMADRES = '';
ELSE
	TXTMADRES = 'yyy@gmail.com , zzz@outlook.com ' ;
ENDIF;

SENDMAIL MESSAGE TXTMAILMESSAGE TOADDRESS TXTMADRES  HOST CLKCLB011TMP_OHOST PASSW CLKCLB011TMP_OPAWSWORD USERNAME CLKCLB011TMP_OUSERNAME FROM CLKCLB011TMP_ADDRESS  SUBJECT CLKBPM012TMP_SUBJECT ENCPRO 'TLS' ;
SHUTDOWN;
